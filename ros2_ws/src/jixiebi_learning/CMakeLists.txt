cmake_minimum_required(VERSION 3.8)
project(jixiebi_learning)

# 只链接ROS和类型定义，不链接native驱动，也不打开串口。
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(jixiebi_interfaces REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(std_msgs REQUIRED)
find_package(orocos_kdl_vendor REQUIRED)
find_package(orocos_kdl REQUIRED)
find_package(kdl_parser REQUIRED)

add_executable(lesson07_service_server src/lesson07_service_server.cpp)
add_executable(lesson07_service_client src/lesson07_service_client.cpp)
add_executable(lesson08_action_server src/lesson08_action_server.cpp)
add_executable(lesson08_action_client src/lesson08_action_client.cpp)
add_executable(lesson09_marker_publisher src/lesson09_marker_publisher.cpp)
add_executable(lesson10_marker_parameters src/lesson10_marker_parameters.cpp)
add_executable(lesson11_tf_broadcaster src/lesson11_tf_broadcaster.cpp)
add_executable(lesson15_trajectory_preview src/lesson15_trajectory_preview.cpp)
add_executable(lesson16_trajectory_player src/lesson16_trajectory_player.cpp)
add_executable(lesson17_follow_server src/lesson17_follow_server.cpp)
add_executable(lesson17_follow_client src/lesson17_follow_client.cpp)
add_executable(lesson18_from_state src/lesson18_from_state.cpp)
add_executable(lesson19_tool_position src/lesson19_tool_position.cpp)
add_executable(lesson20_tool_ik src/lesson20_tool_ik.cpp)
add_executable(lesson21_cartesian_line src/lesson21_cartesian_line.cpp)
add_executable(lesson22_timed_line src/lesson22_timed_line.cpp)
add_executable(lesson23_compare_ramps src/lesson23_compare_ramps.cpp)
set(lesson_targets lesson07_service_server lesson07_service_client
    lesson08_action_server lesson08_action_client
    lesson09_marker_publisher lesson10_marker_parameters lesson11_tf_broadcaster
    lesson15_trajectory_preview lesson16_trajectory_player lesson17_follow_server lesson17_follow_client
    lesson18_from_state lesson19_tool_position lesson20_tool_ik lesson21_cartesian_line lesson22_timed_line
    lesson23_compare_ramps)
foreach(target IN LISTS lesson_targets)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
  ament_target_dependencies(${target} rclcpp jixiebi_interfaces)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
ament_target_dependencies(lesson08_action_server rclcpp_action)
ament_target_dependencies(lesson08_action_client rclcpp_action)
ament_target_dependencies(lesson09_marker_publisher visualization_msgs)
ament_target_dependencies(lesson10_marker_parameters rcl_interfaces visualization_msgs)
ament_target_dependencies(lesson11_tf_broadcaster geometry_msgs tf2_ros)
ament_target_dependencies(lesson15_trajectory_preview rcl_interfaces trajectory_msgs)
ament_target_dependencies(lesson16_trajectory_player trajectory_msgs sensor_msgs)
ament_target_dependencies(lesson17_follow_server control_msgs rcl_interfaces rclcpp_action trajectory_msgs sensor_msgs)
ament_target_dependencies(lesson17_follow_client control_msgs rclcpp_action trajectory_msgs)
ament_target_dependencies(lesson18_from_state control_msgs rclcpp_action sensor_msgs trajectory_msgs)
ament_target_dependencies(lesson19_tool_position sensor_msgs std_msgs kdl_parser orocos_kdl)
ament_target_dependencies(lesson20_tool_ik control_msgs rclcpp_action trajectory_msgs sensor_msgs std_msgs kdl_parser orocos_kdl)
ament_target_dependencies(lesson21_cartesian_line control_msgs rclcpp_action trajectory_msgs sensor_msgs std_msgs kdl_parser orocos_kdl visualization_msgs)
ament_target_dependencies(lesson22_timed_line control_msgs rclcpp_action trajectory_msgs sensor_msgs std_msgs kdl_parser orocos_kdl visualization_msgs)
ament_target_dependencies(lesson23_compare_ramps control_msgs rclcpp_action trajectory_msgs sensor_msgs std_msgs kdl_parser orocos_kdl visualization_msgs)

# ros2 run通过包名和程序名在此安装位置寻找可执行文件。
install(TARGETS ${lesson_targets} DESTINATION lib/${PROJECT_NAME})
# 视角、话题与固定坐标系随包安装，换终端也可从包的share目录找到。
install(DIRECTORY rviz launch urdf config DESTINATION share/${PROJECT_NAME})
ament_package()
